% Code blocks: smaller type, wrap long lines rather than run into the margin.
% baselinestretch is pinned because setspace's \setstretch, which the
% linestretch variable turns on, is global and would otherwise loosen code to
% match the prose leading. Code reads better tighter than prose.
\usepackage{fvextra}
\DefineVerbatimEnvironment{Highlighting}{Verbatim}{
  fontsize=\small,
  baselinestretch=1,
  numbers=left,
  numbersep=8pt,
  xleftmargin=25pt,
  breaklines,
  breakanywhere,
  commandchars=\\\{\}
}

% Make the code block line numbers render with the same monospace font
% as the rest of the code
\renewcommand{\theFancyVerbLine}{%
  \ttfamily\small\color{black!50}\arabic{FancyVerbLine}}

% Start every top-level section on a new page. \clearpage rather than
% \newpage so pending floats flush before the break instead of drifting into
% the following section.
\usepackage{titlesec}
\newcommand{\sectionbreak}{\clearpage}

% Headings in Sans against the Serif body. Starred \titleformat* replaces
% only the font, so the class's own sizes and spacing carry over unchanged.
% The sizes restate the article defaults and must be given per level;
% only the face is shared.
\newcommand{\headingface}{\sffamily}
\titleformat*{\section}{\headingface\Large\bfseries}
\titleformat*{\subsection}{\headingface\large\bfseries}
\titleformat*{\subsubsection}{\headingface\normalsize\bfseries}
\titleformat*{\paragraph}{\headingface\normalsize\bfseries}
\titleformat*{\subparagraph}{\headingface\normalsize\bfseries}

% Inline code, on the shadecolor that pandoc derives from code.theme. \fboxsep
% is tighter than the 3pt default, which otherwise inflates line spacing.
\let\oldtexttt\texttt
\renewcommand{\texttt}[1]{{%
  \setlength{\fboxsep}{1.5pt}%
  \colorbox{shadecolor}{\small\oldtexttt{#1}}%
}}
